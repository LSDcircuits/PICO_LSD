// make the file
mkdir {file name}

// download and check the SDK
git clone https://github.com/raspberrypi/pico-sdk.git
cd pico-sdk
git checkout 2.0.0        # or later; 2.0.0 is the first with RP2350 support
git submodule update --init --recursive

// chekc gcc
arm-none-eabi-gcc --version


// COPY EXTERNAL TO PROJECT FILE

// Cmake project
# 1. Set the SDK path (do this once per terminal session, or put it in your .bashrc)
export PICO_SDK_PATH=/path/to/pico-sdk

# 2. Generate build files (creates the build/ directory automatically)
cmake -S . -B build -DPICO_BOARD=pico2

# 3. Compile
cmake --build build
